cmake_minimum_required(VERSION 3.24)
project(ember_occt_core_contract LANGUAGES CXX)
find_package(OpenCASCADE 8.0.1 REQUIRED)
add_executable(occt-core core.cxx)
add_executable(occt-stacktrace stacktrace.cxx)
foreach(probe occt-core occt-stacktrace)
  target_compile_features(${probe} PRIVATE cxx_std_17)
  target_compile_options(${probe} PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(${probe} SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
  target_link_directories(${probe} PRIVATE ${OpenCASCADE_LIBRARY_DIR})
  target_link_libraries(${probe} PRIVATE TKernel)
  install(TARGETS ${probe} RUNTIME DESTINATION bin)
endforeach()
