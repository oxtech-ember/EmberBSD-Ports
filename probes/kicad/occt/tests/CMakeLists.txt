cmake_minimum_required(VERSION 3.24)
project(ember_occt_core_contract LANGUAGES CXX)
find_package(OpenCASCADE 8.0.1 REQUIRED)
add_executable(occt-core core.cxx)
add_executable(occt-stacktrace stacktrace.cxx)
foreach(probe occt-core occt-stacktrace)
  target_compile_features(${probe} PRIVATE cxx_std_17)
  target_compile_options(${probe} PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(${probe} SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
  target_link_directories(${probe} PRIVATE ${OpenCASCADE_LIBRARY_DIR})
  target_link_libraries(${probe} PRIVATE TKernel)
  install(TARGETS ${probe} RUNTIME DESTINATION bin)
endforeach()

option(EMBERBSD_OCCT_STEP "Build the installed OCCT8 STEP geometry contract" OFF)
option(EMBERBSD_OCCT_STEP_UBSAN "Instrument only the STEP consumer with GCC UBSan traps" OFF)
if(EMBERBSD_OCCT_STEP)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR
      NOT CMAKE_CXX_COMPILER_VERSION MATCHES "^16\\.")
    message(FATAL_ERROR "The STEP consumer requires the common GCC16 compiler")
  endif()
  foreach(toolkit TKPrim TKTopAlgo TKDESTEP)
    if(NOT TARGET ${toolkit})
      message(FATAL_ERROR "The installed OCCT8 package lacks ${toolkit}")
    endif()
  endforeach()
  add_executable(occt-step step.cxx)
  target_compile_features(occt-step PRIVATE cxx_std_17)
  target_compile_options(occt-step PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(occt-step SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
  target_link_libraries(occt-step PRIVATE TKPrim TKTopAlgo TKDESTEP)
  if(EMBERBSD_OCCT_STEP_UBSAN)
    target_compile_options(occt-step PRIVATE -fsanitize=undefined
      -fsanitize-undefined-trap-on-error -fno-sanitize-recover=all)
    target_link_options(occt-step PRIVATE -fsanitize=undefined
      -fsanitize-undefined-trap-on-error -fno-sanitize-recover=all)
  endif()
  install(TARGETS occt-step RUNTIME DESTINATION bin)
endif()
