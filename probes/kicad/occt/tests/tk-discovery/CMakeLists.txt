cmake_minimum_required(VERSION 4.4)
project(occt_tk_discovery NONE)
foreach(input OCCT_SOURCE TCL_PREFIX TK_PREFIX)
  if(NOT IS_ABSOLUTE "${${input}}" OR NOT IS_DIRECTORY "${${input}}")
    message(FATAL_ERROR "Supply an absolute existing ${input}")
  endif()
endforeach()
if(NOT EXISTS "${TCL_PREFIX}/lib/libtcl9.1.so" OR
   NOT EXISTS "${TCL_PREFIX}/include/tcl.h")
  message(FATAL_ERROR "The actual Tcl 9.1 provider is required")
endif()
# Use the explicit providers; never pick a developer machine's older Tk.
set(CMAKE_FIND_USE_CMAKE_SYSTEM_PATH FALSE)
set(CMAKE_FIND_USE_SYSTEM_ENVIRONMENT_PATH FALSE)
set(CMAKE_FIND_USE_CMAKE_ENVIRONMENT_PATH FALSE)
set(CMAKE_FIND_USE_CMAKE_PATH FALSE)
set(BUILD_SHARED_LIBS ON)
set(USE_TK ON)
set(INSTALL_TCL OFF)
set(INSTALL_TK OFF)
set(3RDPARTY_TCL_DIR "${TCL_PREFIX}")
set(3RDPARTY_TCL_LIBRARY_DIR "${TCL_PREFIX}/lib")
set(3RDPARTY_TCL_LIBRARY "${TCL_PREFIX}/lib/libtcl9.1.so" CACHE FILEPATH "")
set(3RDPARTY_TCL_INCLUDE_DIR "${TCL_PREFIX}/include")
set(3RDPARTY_TK_DIR "${TK_PREFIX}")
set(3RDPARTY_TK_LIBRARY_DIR "${TK_PREFIX}/lib")
set(3RDPARTY_TK_INCLUDE_DIR "${TK_PREFIX}/include")
set(CSF_TclLibs tcl8.6)
set(CSF_TclTkLibs tk8.6)
include("${OCCT_SOURCE}/adm/cmake/occt_macros.cmake")
include("${OCCT_SOURCE}/adm/cmake/3rdparty_macro.cmake")
include("${OCCT_SOURCE}/adm/cmake/tcl.cmake")
if(NOT "${3RDPARTY_TCL_LIBRARY_VERSION_WITH_DOT}" STREQUAL "9.1")
  message(FATAL_ERROR "OCCT did not select Tcl 9.1")
endif()
include("${OCCT_SOURCE}/adm/cmake/tk.cmake")
if(EXPECT_MISSING)
  if(USE_TK)
    message(FATAL_ERROR "The baseline unexpectedly retained Tk")
  endif()
  message(STATUS "Baseline reproduced: requested Tk was silently disabled")
else()
  file(REAL_PATH "${TK_PREFIX}/lib/libtcl9tk9.1.so" expected)
  file(REAL_PATH "${3RDPARTY_TK_LIBRARY}" actual)
  if(NOT USE_TK OR NOT actual STREQUAL expected OR
     NOT "${3RDPARTY_TK_LIBRARY_VERSION_WITH_DOT}" STREQUAL "9.1")
    message(FATAL_ERROR "OCCT did not select the exact paired Tk 9.1 provider")
  endif()
  message(STATUS "Selected the exact paired Tcl/Tk 9.1 providers")
endif()
