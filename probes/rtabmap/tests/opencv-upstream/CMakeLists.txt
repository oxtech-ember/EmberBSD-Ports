cmake_minimum_required(VERSION 3.22)
project(ember_opencv_upstream_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
foreach(path OPENCV_SOURCE OPENCV_BUILD OPENCV_PREFIX EIGEN_PREFIX)
  if(NOT IS_ABSOLUTE "${${path}}")
    message(FATAL_ERROR "An absolute ${path} is required")
  endif()
endforeach()
# Keep the provider's language/ABI flags and private allocator-statistics
# definition. The test framework consumes its private exported interface.
file(READ "${OPENCV_BUILD}/compile_commands.json" provider_database)
string(JSON provider_count LENGTH "${provider_database}")
math(EXPR provider_last "${provider_count} - 1")
set(provider_definitions)
set(provider_options)
foreach(index RANGE ${provider_last})
  string(JSON provider_file GET "${provider_database}" ${index} file)
  if(provider_file MATCHES "/modules/core/src/matrix.cpp$")
    string(JSON provider_command GET "${provider_database}" ${index} command)
    separate_arguments(provider_arguments UNIX_COMMAND "${provider_command}")
    list(REMOVE_AT provider_arguments 0)
    set(skip_next FALSE)
    foreach(argument IN LISTS provider_arguments)
      if(skip_next)
        set(skip_next FALSE)
      elseif(argument STREQUAL "-o" OR argument STREQUAL "-I" OR
             argument STREQUAL "-isystem")
        set(skip_next TRUE)
      elseif(argument STREQUAL "-c" OR argument STREQUAL "${provider_file}" OR
             argument MATCHES "^-I" OR argument STREQUAL "-DCVAPI_EXPORTS")
        # Public installed and private fallback header paths are set below.
      elseif(argument MATCHES "^-D")
        string(SUBSTRING "${argument}" 2 -1 definition)
        list(APPEND provider_definitions "${definition}")
      else()
        list(APPEND provider_options "${argument}")
      endif()
    endforeach()
    break()
  endif()
endforeach()
if(NOT provider_definitions)
  message(FATAL_ERROR "The provider's real core compile command is missing")
endif()
find_package(OpenCV 5.0.0 EXACT CONFIG REQUIRED COMPONENTS core imgproc imgcodecs videoio
  NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH PATHS "${OPENCV_PREFIX}/lib/cmake/opencv5")
find_package(Eigen3 5.0.1 EXACT CONFIG REQUIRED NO_DEFAULT_PATH
  NO_CMAKE_FIND_ROOT_PATH PATHS "${EIGEN_PREFIX}/share/eigen3/cmake")
get_target_property(core_location opencv_core IMPORTED_LOCATION_RELEASE)
get_filename_component(resolved_core "${core_location}" REALPATH)
get_filename_component(expected_lib "${OPENCV_PREFIX}/lib" REALPATH)
string(FIND "${resolved_core}" "${expected_lib}/" inside_prefix)
if(NOT inside_prefix EQUAL 0)
  message(FATAL_ERROR "OpenCV core came from outside the installed provider")
endif()
set(ts "${OPENCV_SOURCE}/modules/ts")
set(core_tests "${OPENCV_SOURCE}/modules/core/test")
# This is the same config generated by modules/ts/CMakeLists.txt. No test
# data path is set: the selected test filter uses no external image files.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/opencv_tests_config.hpp"
  "\n#define OPENCV_INSTALL_PREFIX \"${OPENCV_PREFIX}\"\n")
add_executable(opencv-core-upstream
  "${ts}/src/ts.cpp"
  "${ts}/src/ts_arrtest.cpp"
  "${ts}/src/ts_func.cpp"
  "${ts}/src/ts_gtest.cpp"
  "${ts}/src/ts_perf.cpp"
  "${ts}/src/ts_tags.cpp"
  "${core_tests}/test_main.cpp"
  "${core_tests}/test_mat.cpp"
  "${core_tests}/test_operations.cpp")
target_compile_definitions(opencv-core-upstream PRIVATE ${provider_definitions})
target_compile_options(opencv-core-upstream PRIVATE ${provider_options})
set_property(TARGET opencv-core-upstream PROPERTY NO_SYSTEM_FROM_IMPORTED ON)
target_include_directories(opencv-core-upstream BEFORE PRIVATE
  "${OpenCV_INCLUDE_DIRS}"
  "${CMAKE_CURRENT_BINARY_DIR}"
  "${OPENCV_BUILD}"
  "${ts}/include"
  "${OPENCV_SOURCE}/modules/core/include"
  "${OPENCV_SOURCE}/modules/highgui/include")
target_link_libraries(opencv-core-upstream PRIVATE
  opencv_core opencv_imgproc opencv_imgcodecs opencv_videoio Eigen3::Eigen pthread)
