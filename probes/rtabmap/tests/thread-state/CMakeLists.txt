cmake_minimum_required(VERSION 3.22)
project(rtabmap_thread_state_contract LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT IS_ABSOLUTE "${RTABMAP_SOURCE}" OR NOT EXISTS "${RTABMAP_SOURCE}/utilite/src/UThread.cpp")
  message(FATAL_ERROR "An actual RTAB-Map source tree is required")
endif()
# This upstream branch builds utilite before the OpenCV/PCL discovery. It does
# not set ANDROID or change the selected target platform.
set(ANDROID_PREBUILD ON CACHE BOOL "Build the upstream dependency-free utilite branch" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
add_subdirectory("${RTABMAP_SOURCE}" upstream)
find_package(Threads REQUIRED)
add_executable(thread-state ../thread-state.cpp)
target_link_libraries(thread-state PRIVATE rtabmap::utilite Threads::Threads)
target_compile_options(thread-state PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME thread-state COMMAND thread-state)
set_tests_properties(thread-state PROPERTIES TIMEOUT 60)
add_executable(natural-join ../natural-join.cpp)
target_link_libraries(natural-join PRIVATE rtabmap::utilite Threads::Threads)
target_compile_options(natural-join PRIVATE -Wall -Wextra -Werror)
add_test(NAME natural-join COMMAND natural-join)
set_tests_properties(natural-join PROPERTIES TIMEOUT 60)
add_executable(thread-join ../thread-join.cpp)
target_link_libraries(thread-join PRIVATE rtabmap::utilite Threads::Threads)
target_compile_options(thread-join PRIVATE -Wall -Wextra -Werror)
add_test(NAME thread-join COMMAND thread-join)
set_tests_properties(thread-join PROPERTIES TIMEOUT 60)
