cmake_minimum_required(VERSION 3.20)
project(rtabmap_installed_contract LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Eigen3 5.0.1 EXACT REQUIRED CONFIG PATHS "${EIGEN_PREFIX}/share/eigen3/cmake" NO_DEFAULT_PATH)
find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc imgcodecs features calib geometry video photo stitching videoio
    PATHS "${OPENCV_PREFIX}/lib/cmake/opencv5" NO_DEFAULT_PATH)
find_package(PCL 1.15.1 EXACT REQUIRED CONFIG PATHS "${PCL_PREFIX}/share/pcl-1.15" NO_DEFAULT_PATH)
find_package(GTSAM 4.3.0 EXACT REQUIRED CONFIG PATHS "${GTSAM_PREFIX}/lib/cmake/GTSAM" NO_DEFAULT_PATH)
set(SQLite3_INCLUDE_DIR "${SQLITE_PREFIX}/include")
set(SQLite3_LIBRARY "${SQLITE_PREFIX}/lib/libsqlite3.so")
find_package(SQLite3 3.54.0 EXACT REQUIRED)
find_package(RTABMap 0.23.8 EXACT REQUIRED COMPONENTS core utilite
    PATHS "${PROBE_PREFIX}/lib/rtabmap-0.23" NO_DEFAULT_PATH)
find_package(lz4 1.10.0 EXACT REQUIRED CONFIG)
find_package(Threads REQUIRED)
add_executable(rtabmap-contract contract.cpp)
target_link_libraries(rtabmap-contract PRIVATE rtabmap::core SQLite::SQLite3)
find_package(ZLIB REQUIRED)
add_executable(compression-roundtrip compression.cpp)
target_link_libraries(compression-roundtrip PRIVATE rtabmap::core ZLIB::ZLIB)
add_executable(rtabmap-lifecycle lifecycle.cpp)
target_link_libraries(rtabmap-lifecycle PRIVATE rtabmap::core)
add_executable(opencv-foundations ../../robotics-foundations/tests/opencv.cpp)
target_link_libraries(opencv-foundations PRIVATE ${OpenCV_LIBS} Eigen3::Eigen)
target_include_directories(opencv-foundations SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
add_executable(image-codecs-contract ../../media/tests/image-codecs.cpp)
target_link_libraries(image-codecs-contract PRIVATE ${OpenCV_LIBS})
target_include_directories(image-codecs-contract SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
add_executable(thread-state thread-state.cpp)
target_link_libraries(thread-state PRIVATE rtabmap::utilite Threads::Threads)
add_executable(natural-join natural-join.cpp)
target_link_libraries(natural-join PRIVATE rtabmap::utilite Threads::Threads)
add_executable(thread-join thread-join.cpp)
target_link_libraries(thread-join PRIVATE rtabmap::utilite Threads::Threads)
add_executable(private-flann private-flann.cpp)
target_link_libraries(private-flann PRIVATE rtabmap::core LZ4::lz4)
add_executable(private-lz4 private-lz4.c)
target_include_directories(private-lz4 PRIVATE "${LZ4_PREFIX}/include")
target_link_libraries(private-lz4 PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
if(CMAKE_SYSTEM_NAME STREQUAL "NetBSD")
    # The host of threaded C++ DSOs needs pthread loaded at program startup.
    # A plain C-only host crashes at process exit after late pthread unload.
    target_link_options(private-lz4 PRIVATE "-Wl,--no-as-needed")
endif()
if(NOT EXISTS "${OPENCV_SOURCE}/modules/core/include/opencv2/core/utils/plugin_loader.private.hpp")
    message(FATAL_ERROR "Set OPENCV_SOURCE to the exact patched OpenCV source for the private loader regression")
endif()
add_library(opencv-loader-fixture MODULE plugin-fixture.c)
set_target_properties(opencv-loader-fixture PROPERTIES C_VISIBILITY_PRESET hidden)
add_executable(opencv-plugin-loader plugin-loader.cpp)
target_include_directories(opencv-plugin-loader PRIVATE "${OPENCV_SOURCE}/modules/core/include")
target_link_libraries(opencv-plugin-loader PRIVATE opencv_core ${CMAKE_DL_LIBS})
target_compile_options(opencv-plugin-loader PRIVATE -Werror=return-type)
enable_testing()
add_test(NAME persisted-map-localization COMMAND rtabmap-contract "${CMAKE_CURRENT_BINARY_DIR}/map.db")
set_tests_properties(persisted-map-localization PROPERTIES TIMEOUT 180)
add_test(NAME opencv-foundations COMMAND opencv-foundations)
add_test(NAME image-codecs COMMAND image-codecs-contract)
set_tests_properties(opencv-foundations image-codecs PROPERTIES TIMEOUT 60)
add_test(NAME thread-state COMMAND thread-state)
add_test(NAME natural-join COMMAND natural-join)
add_test(NAME thread-join COMMAND thread-join)
set_tests_properties(thread-join PROPERTIES TIMEOUT 120)
add_test(NAME private-flann COMMAND private-flann)
add_test(NAME private-lz4 COMMAND private-lz4 "${PROBE_PREFIX}/lib/librtabmap_core.so"
    "${PCL_PREFIX}/lib/libflann_cpp.so" "${LZ4_PREFIX}/lib/liblz4.so")
add_test(NAME opencv-plugin-loader COMMAND opencv-plugin-loader "$<TARGET_FILE:opencv-loader-fixture>")
set_tests_properties(thread-state natural-join PROPERTIES TIMEOUT 60)
set_tests_properties(private-flann PROPERTIES TIMEOUT 600)
set_tests_properties(private-lz4 opencv-plugin-loader PROPERTIES TIMEOUT 60)
foreach(mode data-vector data-mat data-read rvl-vector rvl-mat)
    add_test(NAME compression-${mode} COMMAND compression-roundtrip ${mode})
    set_tests_properties(compression-${mode} PROPERTIES TIMEOUT 60)
endforeach()
foreach(mode default gtsam)
    add_test(NAME lifecycle-${mode} COMMAND rtabmap-lifecycle ${mode})
    set_tests_properties(lifecycle-${mode} PROPERTIES TIMEOUT 180)
endforeach()
