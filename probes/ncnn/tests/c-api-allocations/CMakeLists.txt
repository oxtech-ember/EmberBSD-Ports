cmake_minimum_required(VERSION 3.20)
project(ncnn_c_api_allocation_controls LANGUAGES CXX)
include(CTest)
find_package(ncnn 20260526 EXACT CONFIG REQUIRED)
set(NCNN_SOURCE_DIR "" CACHE PATH "Patched ncnn 20260526 source tree")
set(NCNN_BASELINE_SOURCE_DIR "" CACHE PATH "Unmodified pinned ncnn source tree")
set(NCNN_CONFIG_DIR "" CACHE PATH "Runtime build directory with generated src headers")
foreach(profile before after)
  if(profile STREQUAL "before")
    set(source "${NCNN_BASELINE_SOURCE_DIR}/src/c_api.cpp")
  else()
    set(source "${NCNN_SOURCE_DIR}/src/c_api.cpp")
  endif()
  if(NOT EXISTS "${source}" OR NOT EXISTS "${NCNN_CONFIG_DIR}/src/platform.h")
    message(FATAL_ERROR "Set all source and generated-config paths")
  endif()
  add_executable(ncnn-c-api-allocations-${profile} allocations.cpp "${source}")
  target_include_directories(ncnn-c-api-allocations-${profile} PRIVATE
    "${NCNN_SOURCE_DIR}/src" "${NCNN_CONFIG_DIR}/src")
  target_link_libraries(ncnn-c-api-allocations-${profile} PRIVATE ncnn)
  target_link_options(ncnn-c-api-allocations-${profile} PRIVATE "-Wl,--wrap=malloc")
  add_executable(ncnn-c-api-lifetime-${profile} lifetime.cpp "${source}")
  target_include_directories(ncnn-c-api-lifetime-${profile} PRIVATE
    "${NCNN_SOURCE_DIR}/src" "${NCNN_CONFIG_DIR}/src")
  target_link_libraries(ncnn-c-api-lifetime-${profile} PRIVATE ncnn)
  add_executable(ncnn-c-api-callbacks-${profile} callbacks.cpp "${source}")
  target_include_directories(ncnn-c-api-callbacks-${profile} PRIVATE
    "${NCNN_SOURCE_DIR}/src" "${NCNN_CONFIG_DIR}/src")
  target_link_libraries(ncnn-c-api-callbacks-${profile} PRIVATE ncnn)
endforeach()
add_test(NAME c-api-allocations-before
  COMMAND ncnn-c-api-allocations-before --baseline)
add_test(NAME c-api-allocations-after COMMAND ncnn-c-api-allocations-after)
add_test(NAME c-api-lifetime-after COMMAND ncnn-c-api-lifetime-after)
add_test(NAME c-api-callbacks-before COMMAND ncnn-c-api-callbacks-before --baseline)
add_test(NAME c-api-callbacks-after COMMAND ncnn-c-api-callbacks-after)
set_tests_properties(c-api-allocations-before c-api-allocations-after
  c-api-lifetime-after c-api-callbacks-before c-api-callbacks-after PROPERTIES TIMEOUT 60)
