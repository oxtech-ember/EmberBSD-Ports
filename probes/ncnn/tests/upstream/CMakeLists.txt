cmake_minimum_required(VERSION 3.20)
project(ncnn_upstream_controls LANGUAGES CXX)

include(CTest)
find_package(ncnn 20260526 EXACT CONFIG REQUIRED)
set(NCNN_SOURCE_DIR "" CACHE PATH "Verified ncnn 20260526 source directory")
if(NOT EXISTS "${NCNN_SOURCE_DIR}/tests/testutil.cpp")
  message(FATAL_ERROR "Set NCNN_SOURCE_DIR to the verified ncnn source tree")
endif()

add_library(ncnn_testutil STATIC "${NCNN_SOURCE_DIR}/tests/testutil.cpp")
target_include_directories(ncnn_testutil PUBLIC
  "${NCNN_SOURCE_DIR}/tests" "${NCNN_SOURCE_DIR}/src"
  "${NCNN_SOURCE_DIR}/src/layer")
target_link_libraries(ncnn_testutil PUBLIC ncnn)

foreach(name c_api paramdict expression relu pooling innerproduct softmax convolution)
  add_executable(ncnn-upstream-${name} "${NCNN_SOURCE_DIR}/tests/test_${name}.cpp")
  target_link_libraries(ncnn-upstream-${name} PRIVATE ncnn_testutil)
  add_test(NAME upstream-${name} COMMAND ncnn-upstream-${name})
  set_tests_properties(upstream-${name} PROPERTIES TIMEOUT 600)
endforeach()

set(NCNN_BASELINE_SOURCE_DIR "" CACHE PATH "Unmodified pinned upstream source directory")
set(NCNN_CONFIG_DIR "" CACHE PATH "Runtime build directory containing src/platform.h and src/ncnn_export.h")
if(NCNN_BASELINE_SOURCE_DIR AND NCNN_CONFIG_DIR)
  find_package(Threads REQUIRED)
  foreach(profile before after)
    if(profile STREQUAL "before")
      set(cpu_source "${NCNN_BASELINE_SOURCE_DIR}/src/cpu.cpp")
    else()
      set(cpu_source "${NCNN_SOURCE_DIR}/src/cpu.cpp")
    endif()
    add_executable(ncnn-cpu-source-${profile} cpu-count.cpp "${cpu_source}")
    target_include_directories(ncnn-cpu-source-${profile} PRIVATE
      "${NCNN_SOURCE_DIR}/src" "${NCNN_CONFIG_DIR}/src")
    target_link_libraries(ncnn-cpu-source-${profile} PRIVATE Threads::Threads)
    if(CMAKE_SYSTEM_NAME STREQUAL "NetBSD")
      target_compile_definitions(ncnn-cpu-source-${profile} PRIVATE _NETBSD_SOURCE)
    endif()
    add_executable(ncnn-cpu-first-use-${profile} cpu-first-use.cpp "${cpu_source}")
    target_include_directories(ncnn-cpu-first-use-${profile} PRIVATE
      "${NCNN_SOURCE_DIR}/src" "${NCNN_CONFIG_DIR}/src")
    target_link_libraries(ncnn-cpu-first-use-${profile} PRIVATE Threads::Threads)
    if(CMAKE_SYSTEM_NAME STREQUAL "NetBSD")
      target_compile_definitions(ncnn-cpu-first-use-${profile} PRIVATE _NETBSD_SOURCE)
    endif()
  endforeach()
  add_test(NAME cpu-source-after COMMAND ncnn-cpu-source-after)
  add_test(NAME cpu-first-use-after COMMAND ncnn-cpu-first-use-after)
  set_tests_properties(cpu-source-after cpu-first-use-after PROPERTIES TIMEOUT 30)
endif()
