cmake_minimum_required(VERSION 3.28)
project(EmberBSDOnnxHeaderReason LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_path(ONNX_RUNTIME_INCLUDE_DIR onnxruntime_c_api.h
  PATH_SUFFIXES onnxruntime REQUIRED)
add_executable(onnx-header-reason-c reason.c)
add_executable(onnx-header-reason-cxx reason.cpp)
foreach(target onnx-header-reason-c onnx-header-reason-cxx)
  target_include_directories(${target} PRIVATE "${ONNX_RUNTIME_INCLUDE_DIR}")
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic-errors)
endforeach()
if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
  target_compile_options(onnx-header-reason-c PRIVATE -Wshift-overflow=2)
endif()
enable_testing()
add_test(NAME onnx-header-reason-c COMMAND onnx-header-reason-c)
add_test(NAME onnx-header-reason-cxx COMMAND onnx-header-reason-cxx)
