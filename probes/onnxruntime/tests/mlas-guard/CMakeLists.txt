cmake_minimum_required(VERSION 3.28)
project(EmberBSDOnnxMlasGuard LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
foreach(required ORT_PRISTINE_SOURCE ORT_SOURCE_DIR ORT_BUILD_DIR GTEST_SOURCE_DIR)
  if(NOT IS_ABSOLUTE "${${required}}" OR NOT IS_DIRECTORY "${${required}}")
    message(FATAL_ERROR "${required} must name an existing absolute directory")
  endif()
endforeach()
find_package(Eigen3 5.0.1 CONFIG REQUIRED)
find_program(RUBY_EXECUTABLE ruby REQUIRED)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
add_subdirectory("${GTEST_SOURCE_DIR}" googletest EXCLUDE_FROM_ALL)
file(READ "${ORT_BUILD_DIR}/compile_commands.json" database)
string(JSON count LENGTH "${database}")
math(EXPR last "${count} - 1")
set(definitions)
foreach(index RANGE 0 ${last})
  string(JSON source GET "${database}" ${index} file)
  if(source MATCHES "/onnxruntime/core/common/threadpool\\.cc$")
    string(JSON command GET "${database}" ${index} command)
    separate_arguments(arguments UNIX_COMMAND "${command}")
    foreach(argument IN LISTS arguments)
      if(argument MATCHES "^-D(.+)$")
        list(APPEND definitions "${CMAKE_MATCH_1}")
      endif()
    endforeach()
    break()
  endif()
endforeach()
if(NOT definitions)
  message(FATAL_ERROR "Production ThreadPool definitions are absent")
endif()
foreach(variant baseline patched)
  if(variant STREQUAL "baseline")
    set(source_root "${ORT_PRISTINE_SOURCE}")
  else()
    set(source_root "${ORT_SOURCE_DIR}")
  endif()
  set(header "${source_root}/onnxruntime/test/mlas/unittest/test_util.h")
  add_executable(mlas-guard-${variant} guard.cpp)
  target_compile_definitions(mlas-guard-${variant} PRIVATE ${definitions}
    MLAS_GUARD_HEADER="${header}")
  target_compile_options(mlas-guard-${variant} PRIVATE -Wall -Wextra -Werror -fno-rtti)
  target_include_directories(mlas-guard-${variant} PRIVATE
    "${ORT_SOURCE_DIR}/include/onnxruntime" "${ORT_SOURCE_DIR}/onnxruntime"
    "${ORT_SOURCE_DIR}/onnxruntime/core/mlas/inc" "${ORT_BUILD_DIR}"
    "${ORT_BUILD_DIR}/_deps/abseil_cpp-src" "${ORT_BUILD_DIR}/_deps/gsl-src/include"
    "${ORT_BUILD_DIR}/_deps/safeint-src" "${ORT_BUILD_DIR}/_deps/flatbuffers-src/include"
    "${ORT_BUILD_DIR}/_deps/onnx-src" "${ORT_BUILD_DIR}/_deps/onnx-build"
    "${ORT_BUILD_DIR}/_deps/protobuf-src/src")
  target_link_libraries(mlas-guard-${variant} PRIVATE GTest::gtest Eigen3::Eigen)
  target_link_options(mlas-guard-${variant} PRIVATE
    -Wl,--wrap=mmap -Wl,--wrap=mprotect -Wl,--wrap=munmap -Wl,--wrap=abort)
endforeach()
enable_testing()
add_test(NAME patched-real COMMAND mlas-guard-patched --real)
add_test(NAME patched-map-failure COMMAND mlas-guard-patched --fail-map)
add_test(NAME patched-guard COMMAND mlas-guard-patched --guard)
add_test(NAME baseline-map-failure COMMAND "${RUBY_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/check-baseline.rb" "$<TARGET_FILE:mlas-guard-baseline>")
