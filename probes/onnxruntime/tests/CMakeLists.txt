cmake_minimum_required(VERSION 3.22)
project(ember_onnx_contracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(onnxruntime 1.31.0 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(previous "${CMAKE_CURRENT_LIST_DIR}/../../ai-engines/tests")
add_executable(onnx-contract "${previous}/onnx.cpp")
target_link_libraries(onnx-contract PRIVATE onnxruntime::onnxruntime)
target_compile_options(onnx-contract PRIVATE -Wall -Wextra -Werror)
if(CMAKE_SYSTEM_NAME STREQUAL "NetBSD")
  add_executable(onnx-affinity-contract "${previous}/onnx-affinity.cpp")
  target_link_libraries(onnx-affinity-contract PRIVATE onnxruntime::onnxruntime Threads::Threads)
  target_compile_options(onnx-affinity-contract PRIVATE -Wall -Wextra -Werror)
endif()
include(CTest)
add_test(NAME onnx-installed-contract COMMAND onnx-contract)
set_tests_properties(onnx-installed-contract PROPERTIES TIMEOUT 60)
if(DEFINED ONNX_TEST_DATA)
  add_executable(silero-contract "${previous}/silero.cpp")
  target_link_libraries(silero-contract PRIVATE onnxruntime::onnxruntime)
  target_compile_options(silero-contract PRIVATE -Wall -Wextra -Werror)
  add_test(NAME onnx-silero-model COMMAND silero-contract
    "${ONNX_TEST_DATA}/silero_vad.onnx" "${ONNX_TEST_DATA}/test.wav")
  set_tests_properties(onnx-silero-model PROPERTIES TIMEOUT 240)
endif()
