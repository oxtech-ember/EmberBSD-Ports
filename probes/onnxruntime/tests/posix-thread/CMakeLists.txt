cmake_minimum_required(VERSION 3.28)
project(EmberBSDOnnxPosixThreadControls LANGUAGES CXX)
find_package(Threads REQUIRED)
find_program(Ruby_EXECUTABLE NAMES ruby REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
foreach(source ORT_PRISTINE_SOURCE ORT_SOURCE_DIR)
  if(NOT IS_ABSOLUTE "${${source}}" OR NOT IS_DIRECTORY "${${source}}")
    message(FATAL_ERROR "${source} must name an existing absolute directory")
  endif()
endforeach()
execute_process(COMMAND "${Ruby_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate-posix-thread-controls.rb"
  "${ORT_PRISTINE_SOURCE}" "${ORT_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/generated"
  COMMAND_ERROR_IS_FATAL ANY)
foreach(variant baseline patched)
  add_executable(posix-thread-${variant} "${CMAKE_BINARY_DIR}/generated/${variant}.cc")
  target_link_libraries(posix-thread-${variant} PRIVATE Threads::Threads)
  target_compile_options(posix-thread-${variant} PRIVATE -Wall -Wextra -Werror)
  if(CMAKE_SYSTEM_NAME STREQUAL "NetBSD")
    target_link_options(posix-thread-${variant} PRIVATE
      "LINKER:--wrap=__libc_thr_attr_init" "LINKER:--wrap=__libc_thr_attr_destroy")
  else()
    target_link_options(posix-thread-${variant} PRIVATE
      "LINKER:--wrap=pthread_attr_init" "LINKER:--wrap=pthread_attr_destroy")
  endif()
  target_link_options(posix-thread-${variant} PRIVATE "LINKER:--wrap=pthread_create")
endforeach()
enable_testing()
if(NOT CMAKE_CROSSCOMPILING)
  add_test(NAME posix-thread-upstream-defects COMMAND "${Ruby_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/check-posix-baseline.rb" "$<TARGET_FILE:posix-thread-baseline>")
endif()
add_test(NAME posix-thread-patched COMMAND posix-thread-patched)
set_tests_properties(posix-thread-patched PROPERTIES TIMEOUT 20)
