cmake_minimum_required(VERSION 3.20)
project(ember_ai_engine_contracts LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(onnxruntime 1.31.0 EXACT REQUIRED CONFIG)
find_package(ncnn 20260526 EXACT REQUIRED CONFIG)
find_package(PkgConfig REQUIRED)
pkg_check_modules(RNNOISE REQUIRED IMPORTED_TARGET rnnoise=0.2)
enable_testing()
add_executable(onnx-contract onnx.cpp)
target_link_libraries(onnx-contract PRIVATE onnxruntime::onnxruntime)
find_package(Threads REQUIRED)
add_executable(onnx-affinity-contract onnx-affinity.cpp)
target_link_libraries(onnx-affinity-contract PRIVATE onnxruntime::onnxruntime Threads::Threads)
target_compile_options(onnx-affinity-contract PRIVATE -Wall -Wextra -Werror)
add_executable(ncnn-contract ncnn.cpp)
target_link_libraries(ncnn-contract PRIVATE ncnn)
add_executable(rnnoise-contract rnnoise.c)
target_link_libraries(rnnoise-contract PRIVATE PkgConfig::RNNOISE m)
add_executable(silero-contract silero.cpp)
target_link_libraries(silero-contract PRIVATE onnxruntime::onnxruntime)
foreach(target onnx-contract ncnn-contract rnnoise-contract silero-contract)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endforeach()
add_test(NAME onnx-contract COMMAND onnx-contract)
add_test(NAME ncnn-contract COMMAND ncnn-contract)
add_test(NAME rnnoise-contract COMMAND rnnoise-contract)
add_test(NAME build-guards COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/check-build-guards.sh" "${CMAKE_CURRENT_SOURCE_DIR}/../build.sh")
add_test(NAME silero-contract COMMAND silero-contract "${AI_DATA}/models/silero_vad.onnx" "${AI_DATA}/fixtures/silero-test.wav")
add_test(NAME silero-invalid-model COMMAND silero-contract "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${AI_DATA}/fixtures/silero-test.wav")
add_test(NAME silero-invalid-wave COMMAND silero-contract "${AI_DATA}/models/silero_vad.onnx" "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
set_tests_properties(silero-invalid-model silero-invalid-wave PROPERTIES WILL_FAIL TRUE TIMEOUT 20)
set_tests_properties(onnx-contract ncnn-contract rnnoise-contract PROPERTIES TIMEOUT 60)
set_tests_properties(silero-contract PROPERTIES TIMEOUT 240)
set_tests_properties(build-guards PROPERTIES TIMEOUT 30)
