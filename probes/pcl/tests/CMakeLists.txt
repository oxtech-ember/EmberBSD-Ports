cmake_minimum_required(VERSION 3.16)
project(pcl_installed_contract LANGUAGES CXX)
enable_testing()
add_executable(pcl-matrix-check matrix-check.cpp)
add_test(NAME pcl-matrix-check COMMAND pcl-matrix-check)
set_tests_properties(pcl-matrix-check PROPERTIES TIMEOUT 5)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(prefix PROBE_PREFIX EIGEN_PREFIX)
  if(NOT IS_ABSOLUTE "${${prefix}}")
    message(FATAL_ERROR "An absolute ${prefix} is required")
  endif()
endforeach()
set(Eigen3_DIR "${EIGEN_PREFIX}/share/eigen3/cmake")
find_package(Eigen3 5.0.1 EXACT CONFIG REQUIRED NO_DEFAULT_PATH PATHS "${Eigen3_DIR}")
target_link_libraries(pcl-matrix-check PRIVATE Eigen3::Eigen)
target_compile_features(pcl-matrix-check PRIVATE cxx_std_17)
set(Boost_DIR /usr/pkg/lib/cmake/Boost-1.91.0)
find_package(Boost 1.91.0 EXACT CONFIG REQUIRED COMPONENTS iostreams)
set(flann_DIR "${PROBE_PREFIX}/lib/cmake/flann")
find_package(flann 1.9.2 EXACT CONFIG REQUIRED NO_DEFAULT_PATH PATHS "${flann_DIR}")
# flann 1.9.2's config does not propagate its include directory to the
# imported target; PCL headers include flann/util/params.h directly.
if(NOT TARGET flann_cpp OR NOT DEFINED flann_INCLUDE_DIRS AND NOT DEFINED FLANN_INCLUDE_DIRS)
  # fallthrough handled below
endif()
include_directories("${PROBE_PREFIX}/include")
set(PCL_SHARED_LIBS ON)
set(PCL_FLANN_REQUIRED_TYPE SHARED)
find_package(PCL 1.15.1 EXACT CONFIG REQUIRED COMPONENTS common filters registration io surface segmentation
  NO_DEFAULT_PATH PATHS "${PROBE_PREFIX}/share/pcl-1.15")
foreach(component common filters registration io surface segmentation)
  get_target_property(location pcl_${component} IMPORTED_LOCATION_RELEASE)
  if(NOT location)
    get_target_property(location pcl_${component} IMPORTED_LOCATION)
  endif()
  string(FIND "${location}" "${PROBE_PREFIX}/lib/" in_prefix)
  if(NOT in_prefix EQUAL 0)
    message(FATAL_ERROR "PCL ${component} came from outside this probe: ${location}")
  endif()
endforeach()
add_executable(pcl-contract contract.cpp)
target_link_libraries(pcl-contract PRIVATE pcl_common pcl_filters pcl_registration)
foreach(case filter empty-filter registration no-correspondence)
  add_test(NAME pcl-${case} COMMAND pcl-contract ${case})
  set_tests_properties(pcl-${case} PROPERTIES TIMEOUT 30)
endforeach()
