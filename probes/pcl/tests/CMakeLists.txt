# Check the selected installation after resolving symlinks. A sibling prefix
# or a library from the build host must not satisfy an installed contract.
function(require_library target prefix)
  set(location "")
  foreach(config RELEASE RELWITHDEBINFO NOCONFIG)
    get_target_property(candidate "${target}" "IMPORTED_LOCATION_${config}")
    if(candidate)
      set(location "${candidate}")
      break()
    endif()
  endforeach()
  if(NOT location)
    get_target_property(location "${target}" IMPORTED_LOCATION)
  endif()
  if(NOT location OR NOT EXISTS "${location}")
    message(FATAL_ERROR "Missing installed library for ${target}: ${location}")
  endif()
  get_filename_component(location "${location}" REALPATH)
  string(FIND "${location}" "${prefix}/lib/" in_prefix)
  if(NOT in_prefix EQUAL 0)
    message(FATAL_ERROR "${target} came from outside ${prefix}: ${location}")
  endif()
endfunction()
