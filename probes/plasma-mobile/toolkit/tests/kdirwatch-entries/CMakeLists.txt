cmake_minimum_required(VERSION 3.30)
project(KDirWatchEntryContract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.12.0 REQUIRED COMPONENTS Core)
qt_standard_project_setup(REQUIRES 6.12)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PCH OFF CACHE BOOL "" FORCE)
set(ENABLE_INOTIFY OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(KCOREADDONS_USE_QML OFF CACHE BOOL "" FORCE)
set(BUILD_PYTHON_BINDINGS OFF CACHE BOOL "" FORCE)
if(CONTRACT_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_subdirectory("${UPSTREAM_SOURCE}" upstream)
set_property(SOURCE "${UPSTREAM_SOURCE}/src/lib/io/kdirwatch.cpp"
  DIRECTORY "${UPSTREAM_SOURCE}/src/lib" APPEND PROPERTY
  COMPILE_OPTIONS -include "${CMAKE_CURRENT_SOURCE_DIR}/netbsd-policy.h")
qt_add_executable(kdirwatch-entry-contract contract.cpp)
target_include_directories(kdirwatch-entry-contract PRIVATE
  "${UPSTREAM_SOURCE}/src/lib/io" "${CMAKE_CURRENT_BINARY_DIR}/upstream/src/lib")
target_link_libraries(kdirwatch-entry-contract PRIVATE KF6::CoreAddons)
