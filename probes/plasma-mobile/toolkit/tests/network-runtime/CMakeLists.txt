cmake_minimum_required(VERSION 3.24)
project(QtNetworkRuntimeCheck LANGUAGES CXX)
find_package(Qt6 6.12 REQUIRED COMPONENTS Network)
add_executable(qt-network-runtime main.cpp)
target_compile_features(qt-network-runtime PRIVATE cxx_std_17)
target_link_libraries(qt-network-runtime PRIVATE Qt6::Network)
enable_testing()
add_test(NAME qt_network_runtime COMMAND qt-network-runtime)
set_tests_properties(qt_network_runtime PROPERTIES TIMEOUT 30)
