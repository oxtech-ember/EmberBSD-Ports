cmake_minimum_required(VERSION 3.22)
project(vtk_pegtl4_compat LANGUAGES CXX)

if(NOT DEFINED VTK_SOURCE OR NOT DEFINED PEGTL_PREFIX)
  message(FATAL_ERROR "Set VTK_SOURCE and PEGTL_PREFIX")
endif()
list(PREPEND CMAKE_MODULE_PATH "${VTK_SOURCE}/CMake")
list(PREPEND CMAKE_PREFIX_PATH "${PEGTL_PREFIX}")
find_package(PEGTL 4.0.2 REQUIRED)
get_target_property(_pegtl_include PEGTL::PEGTL INTERFACE_INCLUDE_DIRECTORIES)
if(NOT _pegtl_include STREQUAL "${PEGTL_PREFIX}/include")
  message(FATAL_ERROR "PEGTL::PEGTL include root is '${_pegtl_include}', expected '${PEGTL_PREFIX}/include'")
endif()
add_executable(vtk_pegtl4_compat ../pegtl4-compat.cpp)
target_compile_features(vtk_pegtl4_compat PRIVATE cxx_std_17)
target_include_directories(vtk_pegtl4_compat PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../include" "${VTK_SOURCE}/IO/MotionFX")
target_link_libraries(vtk_pegtl4_compat PRIVATE PEGTL::PEGTL)
