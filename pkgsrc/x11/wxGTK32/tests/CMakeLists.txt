cmake_minimum_required(VERSION 3.20)
project(wx_webkit6_contract LANGUAGES CXX)
# Use the selected native or target SDK toolchain and installed wx-config.
option(BUILD_WEBKIT_CONTRACT "Build the WebKit6 runtime contract" ON)
option(BUILD_TEXT_CONTRACT "Build the GTK4 text runtime contract" ON)
option(BUILD_MENU_CONTRACT "Build the GTK4 menu runtime contract" ON)
option(BUILD_GL_CONTRACT "Build the GTK4 EGL runtime contract" ON)
set(WX_COMPONENTS core base)
if(BUILD_GL_CONTRACT)
    list(APPEND WX_COMPONENTS gl)
endif()
if(BUILD_WEBKIT_CONTRACT)
    list(APPEND WX_COMPONENTS webview)
endif()
find_package(wxWidgets 3.2 REQUIRED COMPONENTS ${WX_COMPONENTS})
include(${wxWidgets_USE_FILE})
find_package(PkgConfig REQUIRED)
enable_testing()
pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4)
if(BUILD_TEXT_CONTRACT)
add_executable(text4-contract text4-contract.cpp)
target_compile_features(text4-contract PRIVATE cxx_std_11)
target_link_libraries(text4-contract PRIVATE ${wxWidgets_LIBRARIES} PkgConfig::GTK4)
add_test(NAME text4-contract COMMAND text4-contract)
set_tests_properties(text4-contract PROPERTIES TIMEOUT 40)
endif()
if(BUILD_WEBKIT_CONTRACT)
pkg_check_modules(WEBKIT6 REQUIRED IMPORTED_TARGET webkitgtk-6.0)
add_executable(webkit6-contract webkit6-contract.cpp)
target_compile_features(webkit6-contract PRIVATE cxx_std_11)
target_link_libraries(webkit6-contract PRIVATE ${wxWidgets_LIBRARIES} PkgConfig::WEBKIT6)
add_test(NAME webkit6-contract COMMAND webkit6-contract)
set_tests_properties(webkit6-contract PROPERTIES TIMEOUT 40)
endif()

if(BUILD_GL_CONTRACT)
pkg_check_modules(GTK4_GL REQUIRED IMPORTED_TARGET egl>=1.5 epoxy)
add_executable(gl4-contract gl4-contract.cpp)
target_compile_features(gl4-contract PRIVATE cxx_std_11)
target_link_libraries(gl4-contract PRIVATE ${wxWidgets_LIBRARIES} PkgConfig::GTK4 PkgConfig::GTK4_GL)
add_test(NAME gl4-contract COMMAND gl4-contract)
set_tests_properties(gl4-contract PROPERTIES TIMEOUT 20)
endif()

if(BUILD_MENU_CONTRACT)
add_executable(menu4-contract menu4-contract.cpp)
target_compile_features(menu4-contract PRIVATE cxx_std_11)
target_link_libraries(menu4-contract PRIVATE ${wxWidgets_LIBRARIES} PkgConfig::GTK4)
add_test(NAME menu4-contract COMMAND menu4-contract)
set_tests_properties(menu4-contract PROPERTIES TIMEOUT 20)
endif()
