cmake_minimum_required(VERSION 3.20)
project(wx_webkit6_contract LANGUAGES CXX)
# Use the selected native or target SDK toolchain and installed wx-config.
option(BUILD_WEBKIT_CONTRACT "Build the WebKit6 runtime contract" ON)
option(BUILD_NATIVE_SMALL_CONTRACT "Build GTK4 cursor/native widget/MDI contracts" ON)
option(BUILD_OVERLAY_CONTRACT "Build the GTK4 common overlay contract" ON)
option(BUILD_MODAL_CONTRACT "Build the GTK4 native modal/printing contract" ON)
option(BUILD_FILECHOOSER_CONTRACT "Build the GTK4 native file chooser contract" ON)
option(BUILD_ABOUT_CONTRACT "Build the GTK4 about-window runtime contract" ON)
option(BUILD_CALENDAR_CONTRACT "Build the GTK4 common calendar contract" ON)
option(BUILD_DATAVIEW_CONTRACT "Build the GTK4 native DataView contract" ON)
option(BUILD_DND_CONTRACT "Build the interactive GTK4 drag-and-drop fixture" ON)
option(BUILD_CLIPBOARD_CONTRACT "Build the GTK4 clipboard contract" ON)
option(BUILD_TEXT_CONTRACT "Build the GTK4 text runtime contract" ON)
option(BUILD_TOOLBAR_CONTRACT "Build the GTK4 toolbar contract" ON)
option(BUILD_TOPLEVEL_CONTRACT "Build the GTK4 top-level runtime contract" ON)
option(BUILD_NOTEBOOK_CONTRACT "Build the GTK4 notebook/static-box contract" ON)
option(BUILD_STYLE_CONTRACT "Build the GTK4 theme/icon rendering contract" ON)
option(BUILD_MENU_CONTRACT "Build the GTK4 menu runtime contract" ON)
option(BUILD_GL_CONTRACT "Build the GTK4 EGL runtime contract" ON)
set(WX_COMPONENTS core base)
if(BUILD_GL_CONTRACT)
    list(APPEND WX_COMPONENTS gl)
endif()
if(BUILD_WEBKIT_CONTRACT)
    list(APPEND WX_COMPONENTS webview)
endif()
find_package(wxWidgets 3.2 REQUIRED COMPONENTS ${WX_COMPONENTS})
include(${wxWidgets_USE_FILE})
find_package(PkgConfig REQUIRED)
enable_testing()
pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4)
if(BUILD_CALENDAR_CONTRACT)
add_executable(calendar4-contract calendar4-contract.cpp)
target_compile_features(calendar4-contract PRIVATE cxx_std_11)
target_link_libraries(calendar4-contract PRIVATE ${wxWidgets_LIBRARIES})
add_test(NAME calendar4-contract COMMAND calendar4-contract)
set_tests_properties(calendar4-contract PROPERTIES TIMEOUT 20)
endif()
if(BUILD_ABOUT_CONTRACT)
add_executable(about4-contract about4-contract.cpp)
target_compile_features(about4-contract PRIVATE cxx_std_11)

add_test(NAME about4-contract COMMAND about4-contract)
set_tests_properties(about4-contract PROPERTIES TIMEOUT 20)
endif()
if(BUILD_TEXT_CONTRACT)

set_tests_properties(text4-contract PROPERTIES TIMEOUT 40)
endif()

endif()
