+if(PKGSRC_QTTEST_INCLUDE_DIR)
+    # Keep source names useful in installed QML diagnostics.
+    function(pkgsrc_map_qt_source target source_file)
+        if(TARGET ${target})
+            set_property(SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/${source_file}"
+                TARGET_DIRECTORY ${target} APPEND PROPERTY COMPILE_OPTIONS
+                "-fmacro-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}/=src/" ${ARGN})
+        endif()
+    endfunction()
