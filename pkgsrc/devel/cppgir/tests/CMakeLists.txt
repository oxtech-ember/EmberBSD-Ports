cmake_minimum_required(VERSION 3.20)
project(cppgir_consumer LANGUAGES CXX)
find_package(cppgir 2 CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
add_executable(cppgir-consumer main.cpp)
target_compile_definitions(cppgir-consumer PRIVATE GI_INLINE=1)
if(NOT IS_DIRECTORY "${GENERATED_GIR_DIR}")
  message(FATAL_ERROR "Set GENERATED_GIR_DIR to the selected generated GIR headers")
endif()
target_include_directories(cppgir-consumer PRIVATE "${GENERATED_GIR_DIR}")
target_link_libraries(cppgir-consumer PRIVATE CppGir::gi PkgConfig::GIO)
