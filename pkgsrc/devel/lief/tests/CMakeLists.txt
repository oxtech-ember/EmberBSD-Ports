cmake_minimum_required(VERSION 3.16)
project(lief_consumer LANGUAGES CXX)
find_package(LIEF CONFIG REQUIRED)
add_executable(lief_consumer consumer.cpp)
target_compile_features(lief_consumer PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(lief_consumer PRIVATE /UNDEBUG)
else()
  target_compile_options(lief_consumer PRIVATE -UNDEBUG)
endif()
target_link_libraries(lief_consumer PRIVATE LIEF::LIEF)
add_executable(lief_header_link header-link.cpp)
target_compile_features(lief_header_link PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(lief_header_link PRIVATE /UNDEBUG)
else()
  target_compile_options(lief_header_link PRIVATE -UNDEBUG)
endif()
target_link_libraries(lief_header_link PRIVATE LIEF::LIEF)
