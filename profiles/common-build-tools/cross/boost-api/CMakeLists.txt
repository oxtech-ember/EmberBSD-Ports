cmake_minimum_required(VERSION 3.24)
project(EmberInstalledBoost LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "16.2.0")
  message(FATAL_ERROR "Select the common GCC 16.2.0 compiler")
endif()
option(EMBER_BOOST_NATIVE_HEADERS "Build only the native header companion" OFF)
add_executable(boost-api api.cpp)
target_compile_features(boost-api PRIVATE cxx_std_20)
target_compile_options(boost-api PRIVATE -Wall -Wextra -Werror)
if(EMBER_BOOST_NATIVE_HEADERS)
  if(NOT IS_DIRECTORY "${EMBER_BOOST_INCLUDE}/boost")
    message(FATAL_ERROR "Set EMBER_BOOST_INCLUDE to the common installed Boost headers")
  endif()
  find_package(Threads REQUIRED)
  target_compile_definitions(boost-api PRIVATE EMBER_BOOST_NATIVE_HEADERS=1)
  target_include_directories(boost-api PRIVATE "${EMBER_BOOST_INCLUDE}")
  target_link_libraries(boost-api PRIVATE Threads::Threads)
else()
  find_package(Boost 1.92.0 EXACT CONFIG REQUIRED COMPONENTS locale filesystem thread unit_test_framework)
  target_link_libraries(boost-api PRIVATE Boost::locale Boost::filesystem Boost::thread Boost::unit_test_framework)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/loaded.h")
    target_include_directories(boost-api PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  else()
    # Reuse the common public installed-provider inspection contract.
    target_include_directories(boost-api PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../common-graphics/cross/gui-cad-api")
  endif()
  target_link_options(boost-api PRIVATE "LINKER:--no-as-needed")
endif()
