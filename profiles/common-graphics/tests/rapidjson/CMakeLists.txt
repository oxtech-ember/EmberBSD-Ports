cmake_minimum_required(VERSION 3.21)
project(emberbsd_rapidjson_contract LANGUAGES CXX)
# SPDX-License-Identifier: BSD-2-Clause
# Origin: EmberBSD (AI-assisted), installed header API contract.
if(NOT IS_ABSOLUTE "${RAPIDJSON_PREFIX}")
  message(FATAL_ERROR "Set RAPIDJSON_PREFIX to the installed absolute prefix")
endif()
file(REAL_PATH "${RAPIDJSON_PREFIX}" rapidjson_prefix)
find_package(RapidJSON CONFIG REQUIRED PATHS "${rapidjson_prefix}" NO_DEFAULT_PATH)
foreach(path IN LISTS RapidJSON_INCLUDE_DIRS)
  file(REAL_PATH "${path}" actual)
  cmake_path(IS_PREFIX rapidjson_prefix "${actual}" NORMALIZE inside)
  if(NOT inside)
    message(FATAL_ERROR "RapidJSON headers are outside RAPIDJSON_PREFIX")
  endif()
endforeach()
add_executable(rapidjson-contract contract.cpp)
target_compile_features(rapidjson-contract PRIVATE cxx_std_17)
target_link_libraries(rapidjson-contract PRIVATE RapidJSON)
target_compile_options(rapidjson-contract PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME installed-rapidjson COMMAND rapidjson-contract)
