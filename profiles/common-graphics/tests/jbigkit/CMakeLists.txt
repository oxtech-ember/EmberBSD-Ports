cmake_minimum_required(VERSION 3.25)
project(jbigkit_installed_checks C)
enable_testing()
set(JBIG_SOURCE "" CACHE PATH "Patched JBIG-KIT 2.2 source directory")
set(JBIG_PREFIX "" CACHE PATH "Installed or staged JBIG-KIT prefix")
if(NOT IS_ABSOLUTE "${JBIG_SOURCE}" OR NOT IS_ABSOLUTE "${JBIG_PREFIX}")
  message(FATAL_ERROR "JBIG_SOURCE and JBIG_PREFIX must be absolute paths")
endif()
find_path(JBIG_INCLUDE_DIR jbig.h PATHS "${JBIG_PREFIX}/include"
  NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
find_library(JBIG_LIBRARY jbig PATHS "${JBIG_PREFIX}/lib"
  NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
find_library(JBIG85_LIBRARY jbig85 PATHS "${JBIG_PREFIX}/lib"
  NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
file(REAL_PATH "${JBIG_PREFIX}" prefix_real)
foreach(path IN ITEMS "${JBIG_INCLUDE_DIR}/jbig.h" "${JBIG_INCLUDE_DIR}/jbig85.h"
    "${JBIG_LIBRARY}" "${JBIG85_LIBRARY}")
  if(NOT EXISTS "${path}")
    message(FATAL_ERROR "Missing installed provider file: ${path}")
  endif()
  file(REAL_PATH "${path}" resolved)
  cmake_path(IS_PREFIX prefix_real "${resolved}" NORMALIZE within_prefix)
  if(NOT within_prefix)
    message(FATAL_ERROR "Installed provider file is outside JBIG_PREFIX: ${path}")
  endif()
endforeach()
foreach(library IN ITEMS "${JBIG_LIBRARY}" "${JBIG85_LIBRARY}")
  file(REAL_PATH "${library}" resolved)
  if(NOT resolved MATCHES "\.(so(\.[0-9]+)*|dylib)$")
    message(FATAL_ERROR "The installed-library check requires shared libraries")
  endif()
endforeach()
foreach(pair IN ITEMS "jbig.h|JBG" "jbig85.h|JBG85")
  string(REPLACE "|" ";" fields "${pair}")
  list(GET fields 0 header)
  list(GET fields 1 macro)
  file(STRINGS "${JBIG_INCLUDE_DIR}/${header}" version
    REGEX "^#define ${macro}_VERSION[ \t]+\"2\\.2\"$")
  if(NOT version)
    message(FATAL_ERROR "The installed ${header} must be version 2.2")
  endif()
endforeach()
foreach(name IN ITEMS tstcodec tstcodec85 tstjoint)
  # The copy prevents quoted includes from selecting source-tree headers.
  configure_file("${JBIG_SOURCE}/libjbig/${name}.c" "${name}.c" COPYONLY)
  add_executable(${name} "${CMAKE_CURRENT_BINARY_DIR}/${name}.c")
  target_include_directories(${name} PRIVATE "${JBIG_INCLUDE_DIR}")
  target_compile_options(${name} PRIVATE -W -Wall -ansi -pedantic)
  if(name STREQUAL "tstcodec85")
    target_link_libraries(${name} PRIVATE "${JBIG85_LIBRARY}")
  elseif(name STREQUAL "tstcodec")
    target_link_libraries(${name} PRIVATE "${JBIG_LIBRARY}")
  else()
    target_link_libraries(${name} PRIVATE "${JBIG_LIBRARY}" "${JBIG85_LIBRARY}")
  endif()
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 180)
endforeach()
