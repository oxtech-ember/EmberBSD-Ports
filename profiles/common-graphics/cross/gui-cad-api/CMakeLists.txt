cmake_minimum_required(VERSION 3.25)
project(ember_gui_cad_api LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
find_package(glm 1.0.3 EXACT CONFIG REQUIRED)
add_executable(glm-api glm-api.cc)
target_link_options(glm-api PRIVATE "LINKER:--no-as-needed")
target_link_libraries(glm-api PRIVATE glm::glm m)
target_compile_options(glm-api PRIVATE -Wall -Wextra -Werror)
